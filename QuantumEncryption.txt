#### **2. Create the Backend Code**
Create a file named `app.py` in the `backend` folder and add the following code:

```python
from flask import Flask, request, jsonify
from qiskit import QuantumCircuit, Aer, execute
import random

app = Flask(__name__)

# Simulate BB84 protocol
def bb84_key_exchange(length):
    # Alice's random bits and bases
    alice_bits = [random.randint(0, 1) for _ in range(length)]
    alice_bases = [random.choice(['rectilinear', 'diagonal']) for _ in range(length)]

    # Bob's random bases
    bob_bases = [random.choice(['rectilinear', 'diagonal']) for _ in range(length)]

    # Simulate qubit transmission and measurement
    key = []
    for i in range(length):
        qc = QuantumCircuit(1, 1)

        # Alice prepares the qubit
        if alice_bits[i] == 1:
            qc.x(0)  # Apply X gate for |1⟩
        if alice_bases[i] == 'diagonal':
            qc.h(0)  # Apply Hadamard gate for diagonal basis

        # Bob measures the qubit
        if bob_bases[i] == 'diagonal':
            qc.h(0)
        qc.measure(0, 0)

        # Simulate the circuit
        simulator = Aer.get_backend('qasm_simulator')
        result = execute(qc, simulator, shots=1).result()
        measured_bit = int(list(result.get_counts())[0])

        # Compare bases and keep the bit if they match
        if alice_bases[i] == bob_bases[i]:
            key.append(measured_bit)

    return key

# Encrypt a message using the shared key
def encrypt(message, key):
    encrypted = ''.join([str(int(message[i]) ^ key[i % len(key)]) for i in range(len(message))])
    return encrypted

# Decrypt a message using the shared key
def decrypt(encrypted, key):
    decrypted = ''.join([str(int(encrypted[i]) ^ key[i % len(key)]) for i in range(len(encrypted))])
    return decrypted

# API endpoints
@app.route('/generate_key', methods=['GET'])
def generate_key():
    key = bb84_key_exchange(16)  # Generate a 16-bit key
    return jsonify({"key": key})

@app.route('/encrypt', methods=['POST'])
def encrypt_message():
    data = request.json
    message = data['message']
    key = data['key']
    encrypted = encrypt(message, key)
    return jsonify({"encrypted": encrypted})

@app.route('/decrypt', methods=['POST'])
def decrypt_message():
    data = request.json
    encrypted = data['encrypted']
    key = data['key']
    decrypted = decrypt(encrypted, key)
    return jsonify({"decrypted": decrypted})

if __name__ == '__main__':
    app.run(port=5000)
```

#### **3. Run the Backend**
Start the Flask server:
```bash
python app.py
```
The backend will run on `http://localhost:5000`.

### **Step 3: Set Up the Frontend (ReactJS)**

#### **1. Install Node.js and Create React App**
If you don’t have Node.js installed, download it from [nodejs.org](https://nodejs.org/). Then, navigate to the `frontend` folder and create a React app:
```bash
cd ../frontend
npx create-react-app quantum-encryption-poc
cd quantum-encryption-poc
```

#### **2. Install Axios**
Axios is a library for making HTTP requests. Install it:
```bash
npm install axios
```

#### **3. Create the Frontend Code**
Replace the contents of `src/App.js` with the following code:

```javascript
import React, { useState } from "react";
import axios from "axios";

const App = () => {
  const [key, setKey] = useState([]);
  const [message, setMessage] = useState("");
  const [encrypted, setEncrypted] = useState("");
  const [decrypted, setDecrypted] = useState("");

  const generateKey = async () => {
    const response = await axios.get("http://localhost:5000/generate_key");
    setKey(response.data.key);
  };

  const handleEncrypt = async () => {
    const response = await axios.post("http://localhost:5000/encrypt", { message, key });
    setEncrypted(response.data.encrypted);
  };

  const handleDecrypt = async () => {
    const response = await axios.post("http://localhost:5000/decrypt", { encrypted, key });
    setDecrypted(response.data.decrypted);
  };

  return (
    <div style={{ padding: "20px" }}>
      <h1>Quantum Encryption POC</h1>
      <button onClick={generateKey}>Generate Quantum Key</button>
      <p>Key: {key.join("")}</p>

      <input
        type="text"
        value={message}
        onChange={(e) => setMessage(e.target.value)}
        placeholder="Enter a message"
        style={{ margin: "10px", padding: "5px" }}
      />
      <button onClick={handleEncrypt}>Encrypt</button>
      <p>Encrypted: {encrypted}</p>

      <button onClick={handleDecrypt}>Decrypt</button>
      <p>Decrypted: {decrypted}</p>
    </div>
  );
};

export default App;
```

#### **4. Run the Frontend**
Start the React app:
```bash
npm start
```
The frontend will run on `http://localhost:3000`.
